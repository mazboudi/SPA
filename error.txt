variables:
  app_version: "1.1.${CI_PIPELINE_IID}" # Auto-incremented version based on pipeline ID
  nexus_registry: "fmk.nexus-ci.onefiserv.net/apm/0001557"
  image_name: "${nexus_registry}/ocs_authz"
  npm_registry: "https://nexus-cd.onefiserv.net/repository/npm-apm0001557/"
  FMK_DIND_IMG: "fmk.nexus.onefiserv.net/fmk/ext-tools/fmk-dind:FMK-11-21-24"

  # DIND settings (keep as you had them)
  DOCKER_DRIVER: overlay2
  DOCKER_HOST: tcp://docker:2376/
  DOCKER_TLS_CERTDIR: "/certs"
  DOCKER_TLS_VERIFY: "0"
  DOCKER_CERT_PATH: "$DOCKER_TLS_CERTDIR/client"

workflow:
  rules:
    - if: $CI_PIPELINE_SOURCE == 'merge_request_event'
    - if: $CI_COMMIT_BRANCH && $CI_OPEN_MERGE_REQUESTS
      when: never
    - if: $CI_COMMIT_BRANCH

stages:
  - build
  - release

Build Application:
  stage: build
  image: "fmk.nexus.onefiserv.net/fmk/node/node20:FMK-26-03-04"
  script:
    - echo "Configuring npm registry..."
    - npm config set registry "${npm_registry}"
    - npm config set strict-ssl false

    - echo "Installing dependencies..."
    # Using legacy-peer-deps to avoid ERESOLVE peer dependency issues
    - npm install --legacy-peer-deps || (echo "Failed to run npm install." && exit 1)

    - echo "Building application..."
    - npm run build || (echo "Failed to build React application." && exit 1)

    - echo "Build output contents:"
    - ls -la build || true
  artifacts:
    when: always
    expire_in: 1 week
    paths:
      - build/

Build and Push Docker Image:
  stage: release
  image: "$FMK_DIND_IMG"
  services:
    - name: "$FMK_DIND_IMG"
      alias: docker
  tags:
    - docker

  # Make sure artifacts from build job are available before docker build
  needs:
    - job: Build Application
      artifacts: true
  rules:
    - if: $CI_COMMIT_BRANCH == "main"
  script:
    - echo "Logging into Nexus registry..."
    - docker logout fmk.nexus-ci.onefiserv.net || true
    - echo "${CI_REGISTRY_PASSWORD_W}" | docker login fmk.nexus-ci.onefiserv.net --username "${CI_REGISTRY_USER_W}" --password-stdin

    - echo "Building Docker image ${image_name}:${app_version} ..."
    - docker build -t "${image_name}:${app_version}" .

    - echo "Pushing Docker image..."
    - docker push "${image_name}:${app_version}" || (echo "Docker push failed." && exit 1)
